1"""MP3エンコードユーティリティモジュール. 

2 

3numpy 配列の波形データを MP3 ファイルとして保存する機能を提供する。 

4内部的に pydub (ffmpeg wrapper) を使用してエンコードを行う。 

5""" 

6 

7import logging 

8from pathlib import Path 

9 

10import numpy as np 

11 

12logger = logging.getLogger(__name__) 

13 

14DEFAULT_MP3_BITRATE = "128k" 

15 

16 

17def save_as_mp3( 

18 waveform: np.ndarray, 

19 sample_rate: int, 

20 output_path: Path, 

21 bitrate: str = DEFAULT_MP3_BITRATE, 

22) -> None: 

23 """波形データを MP3 ファイルとして保存する. 

24 

25 Args: 

26 waveform: float32 の波形データ（-1.0 〜 1.0） 

27 sample_rate: サンプルレート (Hz) 

28 output_path: 出力先パス（.mp3 拡張子推奨） 

29 bitrate: MP3 ビットレート（例: "128k", "192k", "256k"） 

30 

31 Raises: 

32 ImportError: pydub がインストールされていない場合 

33 RuntimeError: ffmpeg が見つからない場合 

34 """ 

35 from pydub import AudioSegment 

36 

37 # float32 → int16 に変換（クリッピング付き） 

38 clipped = np.clip(waveform, -1.0, 1.0) 

39 int16_data = (clipped * 32767).astype(np.int16) 

40 

41 # AudioSegment を生データから構築 

42 audio = AudioSegment( 

43 data=int16_data.tobytes(), 

44 sample_width=2, # 16-bit 

45 frame_rate=sample_rate, 

46 channels=1, # mono 

47 ) 

48 

49 output_path = Path(output_path) 

50 output_path.parent.mkdir(parents=True, exist_ok=True) 

51 audio.export(str(output_path), format="mp3", bitrate=bitrate) 

52 logger.info("Saved MP3: %s (bitrate=%s)", output_path, bitrate)
